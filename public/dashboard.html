// Send test email function async function sendTestEmail() { const user = window.supabaseClient.currentUser; if (!user) { showError("Please log in to send test email"); Sentry.captureException(new Error("User not logged in when attempting to send test email")); return; } const button = event.target; const originalText = button.textContent; button.textContent = "Sending..."; button.disabled = true; try { const response = await fetch("/api/send-test-email", { method: "POST", headers: { "Content-Type": "application/json", }, body: JSON.stringify({ toEmail: user.email, toName: user.full_name || user.email.split("@")[0], emailType: "test" }) }); const result = await response.json(); if (result.success) { showError('Test email sent successfully!', 'success'); console.log("Email sent:", result); } else { throw new Error(result.error || "Failed to send email"); } } catch (error) { console.error("Error sending test email:", error); Sentry.captureException(error); showError(`Failed to send test email: ${error.message}`); Sentry.captureException(error); } finally { button.textContent = originalText; button.disabled = false; } }
